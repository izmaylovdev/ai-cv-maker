# One-off migration tool (ADR-0021 decision #7). Deliberately not part of any
# app's dependency graph — this runs once, from a developer machine, and is
# deleted along with the Cloud SQL instance after the rollback window closes.
#
#   python3 -m venv .venv && . .venv/bin/activate
#   pip install -r requirements.txt

# psycopg 3. The [binary] extra ships libpq, so there is no local Postgres
# install to match. Crucially it returns numeric columns as Decimal and text
# columns as str — and this tool SELECTs the money columns as ::text, so no
# numeric type is involved in the money path at all.
psycopg[binary]>=3.2,<4

# Firestore client. Reads FIRESTORE_EMULATOR_HOST automatically, which is what
# makes the mandatory emulator rehearsal in README.md a zero-code-change run.
google-cloud-firestore>=2.16,<3
